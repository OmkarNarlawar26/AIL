A-1 — Simple Reflex Agent for Vacuum Cleaner

# Simple Reflex Agent - Vacuum Cleaner

def vacuum_agent(location, status):
    if status == "Dirty":
        return "Suck"
    elif location == "A":
        return "Right"
    elif location == "B":
        return "Left"


# Initial environment
location = "A"
room = {
    "A": "Dirty",
    "B": "Dirty"
}

for step in range(1, 5):
    status = room[location]

    action = vacuum_agent(location, status)

    print("Step:", step)
    print("Location:", location)
    print("Status:", status)
    print("Action:", action)

    if action == "Suck":
        room[location] = "Clean"

    elif action == "Right":
        location = "B"

    elif action == "Left":
        location = "A"

    print("Room:", room)
    print()

A-3 — A* Search for 8-Puzzle


# A* Search for 8 Puzzle

goal = [1, 2, 3,
        4, 5, 6,
        7, 8, 0]


def heuristic(state):

    count = 0

    for i in range(9):

        if state[i] != 0 and state[i] != goal[i]:
            count += 1

    return count


def display(state):

    print(state[0], state[1], state[2])
    print(state[3], state[4], state[5])
    print(state[6], state[7], state[8])
    print()


def astar(start):

    states = [(start, 0)]

    while states:

        # Select state having minimum f = g + h
        states.sort(
            key=lambda x: x[1] + heuristic(x[0])
        )

        state, cost = states.pop(0)

        print("State:")
        display(state)

        if state == goal:
            print("Goal Reached!")
            return

        # Move blank tile
        zero = state.index(0)

        moves = []

        if zero > 2:
            moves.append(zero - 3)  # Up

        if zero < 6:
            moves.append(zero + 3)  # Down

        if zero % 3 != 0:
            moves.append(zero - 1)  # Left

        if zero % 3 != 2:
            moves.append(zero + 1)  # Right

        for position in moves:

            new_state = state.copy()

            new_state[zero], new_state[position] = \
                new_state[position], new_state[zero]

            states.append(
                (new_state, cost + 1)
            )


# Initial state

start = [1, 2, 3,
         4, 0, 6,
         7, 5, 8]


print("Initial State:")
display(start)

astar(start)

B-1 — BFS for 8-Puzzle


# BFS for 8 Puzzle Problem

from collections import deque

goal = (
    1, 2, 3,
    4, 5, 6,
    7, 8, 0
)


def display(state):
    print(state[0], state[1], state[2])
    print(state[3], state[4], state[5])
    print(state[6], state[7], state[8])
    print()


def bfs(start):

    queue = deque()
    queue.append((start, []))

    visited = set()
    visited.add(start)

    while queue:

        state, path = queue.popleft()

        if state == goal:
            return path + [state]

        zero = state.index(0)

        moves = []

        if zero >= 3:
            moves.append(zero - 3)

        if zero < 6:
            moves.append(zero + 3)

        if zero % 3 != 0:
            moves.append(zero - 1)

        if zero % 3 != 2:
            moves.append(zero + 1)

        for position in moves:

            new_state = list(state)

            new_state[zero], new_state[position] = \
                new_state[position], new_state[zero]

            new_state = tuple(new_state)

            if new_state not in visited:

                visited.add(new_state)

                queue.append(
                    (new_state, path + [state])
                )

    return None


# Initial State
start = (
    1, 2, 3,
    4, 0, 6,
    7, 5, 8
)


print("Initial State:")
display(start)

print("Goal State:")
display(goal)


solution = bfs(start)


if solution:

    print("Solution using BFS:")

    for i, state in enumerate(solution):

        print("Step", i)
        display(state)

    print("Total moves:", len(solution) - 1)

else:

    print("No solution found.")

# Tower of Hanoi using State Space Search

def hanoi(n, source, auxiliary, destination):

    if n == 1:
        print("Move disk 1 from", source, "to", destination)
        return

    hanoi(n - 1, source, destination, auxiliary)

    print("Move disk", n, "from", source, "to", destination)

    hanoi(n - 1, auxiliary, source, destination)


n = 3

print("Initial State: (123, , )")
print("Goal State: ( , , 123)")
print("Moves:")

hanoi(n, "Source", "Auxiliary", "Destination")

print("Total moves:", 2 ** n - 1)

A-7 — Local Beam Search for TSP


# Local Beam Search for TSP

def distance(tour, graph):
    total = 0

    for i in range(len(tour) - 1):
        total += graph[tour[i]][tour[i + 1]]

    total += graph[tour[-1]][tour[0]]

    return total


def successors(tour):
    result = []

    for i in range(1, len(tour)):
        for j in range(i + 1, len(tour)):

            new_tour = tour.copy()

            new_tour[i], new_tour[j] = \
                new_tour[j], new_tour[i]

            result.append(new_tour)

    return result


def beam_search(graph, beam, k):

    for _ in range(10):

        all_tours = []

        # Generate successors of all k tours
        for tour in beam:
            all_tours += successors(tour)

        # Sort according to distance
        all_tours.sort(
            key=lambda x: distance(x, graph)
        )

        # Select best k tours
        beam = all_tours[:k]

    # Return best tour
    best = min(
        beam,
        key=lambda x: distance(x, graph)
    )

    return best


# Distance matrix
graph = {
    'A': {'A': 0, 'B': 10, 'C': 15, 'D': 20},
    'B': {'A': 10, 'B': 0, 'C': 35, 'D': 25},
    'C': {'A': 15, 'B': 35, 'C': 0, 'D': 30},
    'D': {'A': 20, 'B': 25, 'C': 30, 'D': 0}
}


# Initial k candidate tours
beam = [
    ['A', 'B', 'C', 'D'],
    ['A', 'B', 'D', 'C'],
    ['A', 'C', 'B', 'D']
]

k = 3


# Run Local Beam Search
best_tour = beam_search(graph, beam, k)

print("Best Tour:", best_tour + ['A'])
print("Minimum Distance:", distance(best_tour, graph))
